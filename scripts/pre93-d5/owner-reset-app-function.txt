  function initOwnerPasswordReset(){
    const requestForm=$('#ownerPasswordResetRequestForm'); if(!requestForm)return;
    const requestResult=$('#ownerPasswordResetRequestResult'); const emergencyPanel=$('#ownerEmergencyRecovery'); const confirmPanel=$('#ownerPasswordResetConfirm');
    const hashParams=new URLSearchParams(String(location.hash||'').replace(/^#/,'')); const token=hashParams.get('reset_token')||'';
    if(token&&confirmPanel){confirmPanel.hidden=false;const tokenInput=confirmPanel.querySelector('input[name="token"]');if(tokenInput)tokenInput.value=token;confirmPanel.scrollIntoView({block:'start'});}
    fetch('/api/owner/auth/status').then(r=>r.json()).then(status=>{if(emergencyPanel)emergencyPanel.hidden=!status.emergencyRecoveryAvailable;const note=$('#ownerPasswordResetAvailability');if(note&&!status.emailPasswordResetConfigured)note.textContent=status.emergencyRecoveryAvailable?'Reset email is not configured yet. You can use the one-time Render recovery option below.':'Reset email is not configured yet. A one-time Render recovery code must be configured before emergency recovery can be used.';}).catch(()=>{});
    requestForm.addEventListener('submit',async event=>{event.preventDefault();requestResult.textContent='Checking owner recovery options…';const body=Object.fromEntries(new FormData(requestForm).entries());const response=await fetch('/api/owner/auth/password-reset/request',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(body)}).then(r=>r.json()).catch(error=>({ok:false,error:error.message}));requestResult.className=response.ok?'result-panel success':'result-panel error';requestResult.textContent=response.ok?(response.emailDeliveryConfigured?'If the owner address matches, check that inbox for a reset link.':(response.emergencyRecoveryAvailable?'Email reset is not configured yet. Use the one-time Render recovery option below.':'Email reset is not configured yet. Configure a one-time Render recovery code to continue.')):(response.error||'Owner password recovery could not start.');if(emergencyPanel&&response.emergencyRecoveryAvailable)emergencyPanel.hidden=false;});
    $('#ownerPasswordResetConfirmForm')?.addEventListener('submit',async event=>{event.preventDefault();const form=event.currentTarget;const result=$('#ownerPasswordResetConfirmResult');const data=Object.fromEntries(new FormData(form).entries());if(data.password!==data.confirmPassword){result.className='result-panel error';result.textContent='The new passwords do not match.';return;}delete data.confirmPassword;const response=await fetch('/api/owner/auth/password-reset/confirm',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(data)}).then(r=>r.json()).catch(error=>({ok:false,error:error.message}));result.className=response.ok?'result-panel success':'result-panel error';result.innerHTML=response.ok?'Password reset complete. <a href="/owner-login.html?next=%2Fcontrol-center.html">Sign in with your new password</a>.':escapeHtml(response.error||'The reset could not be completed.');});
    $('#ownerEmergencyRecoveryForm')?.addEventListener('submit',async event=>{event.preventDefault();const form=event.currentTarget;const result=$('#ownerEmergencyRecoveryResult');const data=Object.fromEntries(new FormData(form).entries());if(data.password!==data.confirmPassword){result.className='result-panel error';result.textContent='The new passwords do not match.';return;}delete data.confirmPassword;const response=await fetch('/api/owner/auth/password-reset/emergency',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(data)}).then(r=>r.json()).catch(error=>({ok:false,error:error.message}));result.className=response.ok?'result-panel success':'result-panel error';result.innerHTML=response.ok?'Owner password reset complete. <a href="/owner-login.html?next=%2Fcontrol-center.html">Sign in now and enable MFA</a>.':escapeHtml(response.error||'The one-time recovery reset could not be completed.');});
  }
