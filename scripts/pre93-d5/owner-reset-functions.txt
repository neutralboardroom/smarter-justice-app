async function requestPasswordReset(addressValue){
  const state=readState(); const address=security.normalizeEmail(addressValue); const account=state.accounts.find(item=>item.email===address && item.status==='active');
  state.passwordResetRequests=state.passwordResetRequests.filter(item=>Date.parse(item.expiresAt)>Date.now()&&!item.usedAt);
  if(!account){ writeState(state); return {accepted:true,deliveryAvailable:mailer.status().configured}; }
  state.passwordResetRequests=state.passwordResetRequests.filter(item=>item.accountId!==account.id);
  const token=store.secureToken('ownerreset'); const createdAt=now(); const expiresAt=new Date(Date.now()+30*60000).toISOString();
  state.passwordResetRequests.push({id:store.uid('ownerreset',8),accountId:account.id,tokenHash:security.hashToken(token),createdAt,expiresAt,usedAt:''}); writeState(state);
  const base=String(process.env.APP_BASE_URL||'').replace(/\/$/,''); const resetLink=`${base}/owner-password-reset.html#reset_token=${encodeURIComponent(token)}`;
  let delivery={sent:false,reason:'delivery-error'}; try{delivery=await mailer.sendNotification({kind:'owner_password_reset',to:account.email,payload:{message:'A password reset was requested for the Smarter Justice owner account.',actionLabel:'Reset owner password',actionLink:resetLink}});}catch{}
  store.addAudit({actor:'owner-account',action:'owner_password_reset_requested',details:{accountId:account.id,delivery:delivery.sent?'sent':security.clean(delivery.reason||'not-sent',80)}});
  return {accepted:true,deliveryAvailable:mailer.status().configured,...(process.env.NODE_ENV==='test'?{testToken:token}:{})};
}
function resetPassword(tokenValue,password){
  const state=readState(); const tokenHash=security.hashToken(String(tokenValue||'')); const request=state.passwordResetRequests.find(item=>item.tokenHash===tokenHash&&!item.usedAt&&Date.parse(item.expiresAt)>Date.now());
  if(!request)return {error:'This owner password-reset link is invalid or expired.'};
  if(String(password||'').length<PASSWORD_MIN_LENGTH)return {error:`Use a password with at least ${PASSWORD_MIN_LENGTH} characters.`};
  const account=state.accounts.find(item=>item.id===request.accountId&&item.status==='active'); if(!account)return {error:'This owner account is not active.'};
  const pw=security.hashPassword(password); account.passwordSalt=pw.salt; account.passwordHash=pw.hash; account.updatedAt=now(); for(const item of state.passwordResetRequests)if(item.accountId===account.id)item.usedAt=now(); state.sessions=state.sessions.filter(item=>item.accountId!==account.id); writeState(state);
  store.addAudit({actor:'owner-account',action:'owner_password_reset_completed',details:{accountId:account.id,method:'email-token'}}); return {account:publicAccount(account)};
}
function emergencyResetPassword(input={}){
  const configured=String(process.env.OWNER_ACCOUNT_RECOVERY_CODE||'').trim(); if(configured.length<24)return {error:'One-time owner recovery is not configured.'};
  const supplied=String(input.recoveryCode||'').trim(); const configuredHash=security.hashToken(configured); const suppliedHash=security.hashToken(supplied);
  if(!security.timingSafeTextEqual(configuredHash,suppliedHash))return {error:'The one-time owner recovery code was not accepted.'};
  const state=readState(); if(state.emergencyRecoveryUses.some(item=>item.codeHash===configuredHash))return {error:'That one-time owner recovery code has already been used.'};
  const address=security.normalizeEmail(input.email); const account=state.accounts.find(item=>item.email===address&&item.status==='active'); if(!account)return {error:'The owner email or recovery code was not accepted.'};
  if(String(input.password||'').length<PASSWORD_MIN_LENGTH)return {error:`Use a password with at least ${PASSWORD_MIN_LENGTH} characters.`};
  const pw=security.hashPassword(input.password); account.passwordSalt=pw.salt; account.passwordHash=pw.hash; account.updatedAt=now(); state.sessions=state.sessions.filter(item=>item.accountId!==account.id); for(const item of state.passwordResetRequests)if(item.accountId===account.id)item.usedAt=now(); state.emergencyRecoveryUses.push({codeHash:configuredHash,accountId:account.id,usedAt:now()}); writeState(state);
  store.addAudit({actor:'owner-account',action:'owner_password_reset_completed',details:{accountId:account.id,method:'render-one-time-recovery'}}); return {account:publicAccount(account)};
}
function ownerRecoveryStatus(){
  const state=readState(); const configured=String(process.env.OWNER_ACCOUNT_RECOVERY_CODE||'').trim(); const codeHash=configured?security.hashToken(configured):''; const consumed=Boolean(codeHash&&state.emergencyRecoveryUses.some(item=>item.codeHash===codeHash));
  return {passwordResetFoundation:true,emailDeliveryConfigured:mailer.status().configured,emergencyRecoveryConfigured:configured.length>=24,emergencyRecoveryAvailable:configured.length>=24&&!consumed};
}
